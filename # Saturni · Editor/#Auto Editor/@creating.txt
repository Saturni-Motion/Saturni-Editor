import xml.etree.ElementTree as ET

def create_premiere_xml(file_path, segments, output_xml)
    
    file_path путь к исходному видео
    segments список кортежей [(start_sec, end_sec), ...] которые ИИ решил оставить
    
    duration_frames = 100000  # Примерная длительность
    
    xmeml = ET.Element(xmeml, version=5)
    project = ET.SubElement(xmeml, project)
    name = ET.SubElement(project, name)
    name.text = AI_Rough_Cut
    
    children = ET.SubElement(project, children)
    sequence = ET.SubElement(children, sequence)
    seq_name = ET.SubElement(sequence, name)
    seq_name.text = Clean_Material
    
    media = ET.SubElement(sequence, media)
    video = ET.SubElement(media, video)
    track = ET.SubElement(video, track)
    
    current_out = 0
    for i, (start, end) in enumerate(segments)
        # Переводим секунды в кадры (допустим 30fps)
        start_f = int(start  30)
        end_f = int(end  30)
        clip_duration = end_f - start_f
        
        clipitem = ET.SubElement(track, clipitem, id=fclipitem_{i})
        ET.SubElement(clipitem, name).text = Source_Video
        ET.SubElement(clipitem, duration).text = str(duration_frames)
        ET.SubElement(clipitem, start).text = str(current_out)
        ET.SubElement(clipitem, end).text = str(current_out + clip_duration)
        ET.SubElement(clipitem, in).text = str(start_f)
        ET.SubElement(clipitem, out).text = str(end_f)
        
        file = ET.SubElement(clipitem, file, id=file_1)
        ET.SubElement(file, pathurl).text = ffilelocalhost{file_path.replace('', '')}
        
        current_out += clip_duration

    tree = ET.ElementTree(xmeml)
    tree.write(output_xml, encoding=UTF-8, xml_declaration=True)
    print(fProject saved to {output_xml})

Минусы Gling.ai в сравнении с твоей идеей
Жесткая логика «Лучшего дубля» Gling обычно просто ищет последний дубль одной и той же фразы, считая его удачным. Он не знает, что если ты сказал «Так, это был брак по свету, откатываемся на две фразы назад», то нужно вырезать конкретный кусок. Отсутствие поддержки кастомных тегов На съемках часто используют специфические маркеры хлопки, кодовые слова, фразы вроде «Вариант А» и «Вариант Б». Gling их не поймет как команды. Твой ИИ-скрипт может интерпретировать транскрипт как лог команд, превращая слова в действия по монтажу.

Облачная зависимость и цена Gling — это SaaS (софт как услуга). Ты платишь за часы обработки или подписку. Твое решение на Python + Whisper + API (или локальная LLM) стоит копейки или вообще бесплатно, работая на твоем железе.

Приватность Для многих продакшнов заливать исходники (даже если только звук) в облако — это риск. Локальный скрипт гарантирует, что материал не уйдет за пределы твоей машины.

class PremiereProjectGenerator
    def __init__(self, source_path, fps=30)

        self.fps = fps

    def generate_xml(self, segments, output_path)
        xmeml = ET.Element(xmeml, version=5)
        sequence = ET.SubElement(xmeml, sequence)
        ET.SubElement(sequence, name).text = AI_Rough_Cut_Result
        
        media = ET.SubElement(sequence, media)
        video = ET.SubElement(media, video)
        track = ET.SubElement(video, track)
        
        curr_timeline_pos = 0
        for i, seg in enumerate(segments)
            start_frame = int(seg['start']  self.fps)
            end_frame = int(seg['end']  self.fps)
            duration = end_frame - start_frame
            
            clip = ET.SubElement(track, clipitem, id=fclip_{i})
            ET.SubElement(clip, name).text = Source_Clip
            ET.SubElement(clip, start).text = str(curr_timeline_pos)

            ET.SubElement(clip, out).text = str(end_frame)

            ET.SubElement(file, pathurl).text = ffilelocalhost{self.source_path.replace('', '')}
            
            curr_timeline_pos += duration
            
        tree = ET.ElementTree(xmeml)
        tree.write(output_path, encoding=UTF-8, xml_declaration=True)
